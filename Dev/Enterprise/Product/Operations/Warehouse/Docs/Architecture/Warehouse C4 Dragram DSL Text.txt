        pwhForkliftDriver -> pwhRfApp "Commences pick/put-away & scans packages being put away inside the warehouse or picked from the warehouse to the dock door"
        pwhCW1 -> pwhClient "Sends an email to notify completed receive jobs or orders"
        pwhClient -> pwhWebTracker "Records an upcoming receive jobs (ASN) or dispatch jobs (orders)"
        pwhAdmin -> pwhCW1 "Manually records upcoming receive jobs (ASN) or dispatch jobs (orders)"

        cyYardPlanner -> cyWebApp "Updates inventory, determines yard storage"

        gateIntegration -> dock "Requests a dock to be assigned for a vehicle"
        gateIntegration -> booking "Sends gate-in & gate-out actuals including unannounced"
        gateIntegration -> twhIntegration "Sends gate-in and gate-out events"
        gateIntegration -> cyIntegration "Sends gate-in and gate-out events"
        twhIntegration -> dock "Sends dock status update events"
        forwarding -> twhIntegration "Sends dispatch or receive instructions"
        twhIntegration -> forwarding "Sends updates on dispatch or receive task progress & package level data"
        pwhCW1 -> pwh3rdParty "Sends a notification through eHub to notify compelted receive jobs or orders"
        pwh3rdParty -> pwhCW1 "Sends a notification through eHub to record upcoming receive jobs (ASN) or dispatch jobs (orders)"
        forwarding -> pwhCW1 "Notifies product warehouse of customs declaration (becomes an ASN) or shipments (becomes an order)"
        shippingLine -> cyIntegration "Records authority to release or drop-off"
        cyIntegration -> shippingLine "Sends gate-in & gate-out events"
        landTransport -> cyIntegration "Sends transport run sheets"
