        localFile2 = this.fc.getSelectedFile();

        SwingUtilities.invokeLater(new Runnable()
        {
          public void run()
          {
            UIManager.put("swing.boldMetal", Boolean.FALSE);
            CarXmlConverter.this.createAndShowGUI2(localFile2);

        });
      }
      else
      {
        this.log.append("Open command cancelled by user.\n");
      }
      this.log.setCaretPosition(this.log.getDocument().getLength());
    }
    else
    {
      Object localObject;
      if (paramActionEvent.getSource() == this.checkButton)
      {
        this.fc = new JFileChooser();
        str1 = this.PFR.readParaFile("CARXMLSOURCE");

        this.fc.addChoosableFileFilter(new xmlFilter());
        this.fc.setAcceptAllFileFilterUsed(false);
        
        int j = this.fc.showOpenDialog(this);
        if (j == 0)
        {
          localFile2 = this.fc.getSelectedFile();

          localObject = new XmlCar();
          ((XmlCar)localObject).XmlCarCheck(localFile2);

          } else {
            this.log.append("Failed !! Please check log\n");
          }
        }
        else
        {
          this.log.append("Ckeck command cancelled by user.\n");
        }
        this.log.setCaretPosition(this.log.getDocument().getLength());
      }

      {
        this.fc = new JFileChooser();
        str1 = this.PFR.readParaFile("CARXMLSOURCE");
        localFile1 = new File(str1);
        this.fc.setCurrentDirectory(localFile1);

            this.log.append("Success !! \n");

        }

  protected static ImageIcon createImageIcon(String paramString)
  {
    URL localURL = CarXmlConverter.class.getResource(paramString);

    }
    System.err.println("Couldn't find file: " + paramString);
    return null;
  }

  private void createAndShowGUI2(File paramFile)
  {
    this.log2 = new JTextArea(40, 50);
    this.log2.setMargin(new Insets(5, 5, 5, 5));
    this.log2.setEditable(false);
    JScrollPane localJScrollPane = new JScrollPane(this.log2);
    add(localJScrollPane, "Center");
    try
    {
      FileReader localFileReader = new FileReader(paramFile);
      BufferedReader localBufferedReader = new BufferedReader(localFileReader);
      this.record = new String();
      while ((this.record = localBufferedReader.readLine()) != null)
      {
        this.log2.append(this.record);
        this.log2.append("\n");
      }
    }
    catch (IOException localIOException)
    {
      System.out.println(localIOException);
      this.log2.append("Error..: " + localIOException);
      JOptionPane.showMessageDialog(null, "IO Error :" + localIOException);
    }
    JFrame localJFrame = new JFrame("Car - Xml Converter Log");
    localJFrame.setBounds(30, 30, 0, 0);
    
    localJFrame.setDefaultCloseOperation(1);
    

    localJFrame.getContentPane().add(localJScrollPane);
    

    localJFrame.pack();
    localJFrame.setVisible(true);
  }
