These Integration tests are run with the Infrastructure XML data(system data which excludes the user Routing rule data from eHubPortal) imported from ehubtransactions.db.wisegrid.netm. You can using the below SQL script to populate data from syd-thub-1 or your local as well if you change database's data.
If you want to test with the real database to see if a functiontality works, please use TestRealDatabase/TestRules.cs.

1/ Query to gerate TestDataBase (Please update your TestDataBase/TestData/eHubTransactions_Infrastructure.xml):

DECLARE @FPM_PK uniqueidentifier = '44A7E26D-F2B7-4905-81F0-40BB64D4CE1B'
DECLARE @RoutingRuleRoot uniqueidentifier = (SELECT TOP 1 CC_RR FROM eHubClient WHERE CC_PK = @FPM_PK)

DECLARE @eHubRoutingRule TABLE
(
	[RR_PK] [uniqueidentifier] NOT NULL,
	[RR_Condition_Expression] [nvarchar](max) NULL,
	[RR_Group_RR_GroupRule] [uniqueidentifier] NULL,
	[RR_Group_MatchMultiple] [bit] NULL,
	[RR_Success_CC_Recipient] [uniqueidentifier] NULL,
	[RR_Success_RR_SubRule] [uniqueidentifier] NULL,
	[RR_Failed_ErrorCode] [nvarchar](50) NULL,
	[RR_Failed_ErrorDescription] [nvarchar](2000) NULL,
	[RR_Failed_RR_SubRule] [uniqueidentifier] NULL,
	[RR_Result_Value] [nvarchar](2000) NULL,
	[RR_Success_SP_Provider] [uniqueidentifier] NULL,
	[RR_LastUpdateUTC] [datetime2](7) NULL,
	[RR_Group_Ordering] [int] NULL
)

;WITH x (RR_PK,	RR_Condition_Expression,	RR_Group_RR_GroupRule,	RR_Group_MatchMultiple,	RR_Success_CC_Recipient,	RR_Success_RR_SubRule,	RR_Failed_ErrorCode,	RR_Failed_ErrorDescription,	RR_Failed_RR_SubRule,	RR_Result_Value,	RR_Success_SP_Provider,	RR_LastUpdateUTC,	RR_Group_Ordering)
 AS
 (
   select RR_PK,	RR_Condition_Expression,	RR_Group_RR_GroupRule,	RR_Group_MatchMultiple,	RR_Success_CC_Recipient,	RR_Success_RR_SubRule,	RR_Failed_ErrorCode,	RR_Failed_ErrorDescription,	RR_Failed_RR_SubRule,	RR_Result_Value,	RR_Success_SP_Provider,	RR_LastUpdateUTC,	RR_Group_Ordering
   from eHubRoutingRule
   where RR_PK = @RoutingRuleRoot
   UNION ALL
   select t.RR_PK,	t.RR_Condition_Expression,	t.RR_Group_RR_GroupRule,	t.RR_Group_MatchMultiple,	t.RR_Success_CC_Recipient,	t.RR_Success_RR_SubRule,	t.RR_Failed_ErrorCode,	t.RR_Failed_ErrorDescription,	t.RR_Failed_RR_SubRule,	t.RR_Result_Value,	t.RR_Success_SP_Provider,	t.RR_LastUpdateUTC,	t.RR_Group_Ordering
   from eHubRoutingRule as t inner join
       x as parent on (t.RR_Group_RR_GroupRule = parent.RR_PK OR t.RR_PK = parent.RR_Success_RR_SubRule OR t.RR_PK = parent.RR_Failed_RR_SubRule)
 )
insert into @eHubRoutingRule SELECT * from x ORDER BY RR_Group_RR_GroupRule, RR_Group_Ordering

;WITH x (RR_PK,	RR_Condition_Expression,	RR_Group_RR_GroupRule,	RR_Group_MatchMultiple,	RR_Success_CC_Recipient,	RR_Success_RR_SubRule,	RR_Failed_ErrorCode,	RR_Failed_ErrorDescription,	RR_Failed_RR_SubRule,	RR_Result_Value,	RR_Success_SP_Provider,	RR_LastUpdateUTC,	RR_Group_Ordering)
 AS
 (
   select RR_PK,	RR_Condition_Expression,	RR_Group_RR_GroupRule,	RR_Group_MatchMultiple,	RR_Success_CC_Recipient,	RR_Success_RR_SubRule,	RR_Failed_ErrorCode,	RR_Failed_ErrorDescription,	RR_Failed_RR_SubRule,	RR_Result_Value,	RR_Success_SP_Provider,	RR_LastUpdateUTC,	RR_Group_Ordering
   from eHubRoutingRule
   where RR_PK IN (SELECT RX_RR_ComputeRule FROM eHubRoutingRuleFact WHERE RX_RR = @RoutingRuleRoot AND RX_RR_ComputeRule IS NOT NULL)
   UNION ALL
   select t.RR_PK,	t.RR_Condition_Expression,	t.RR_Group_RR_GroupRule,	t.RR_Group_MatchMultiple,	t.RR_Success_CC_Recipient,	t.RR_Success_RR_SubRule,	t.RR_Failed_ErrorCode,	t.RR_Failed_ErrorDescription,	t.RR_Failed_RR_SubRule,	t.RR_Result_Value,	t.RR_Success_SP_Provider,	t.RR_LastUpdateUTC,	t.RR_Group_Ordering
   from eHubRoutingRule as t inner join
       x as parent on t.RR_Group_RR_GroupRule = parent.RR_PK OR t.RR_PK = parent.RR_Success_RR_SubRule OR t.RR_PK = parent.RR_Failed_RR_SubRule
 )
insert into @eHubRoutingRule SELECT * from x ORDER BY RR_Group_RR_GroupRule, RR_Group_Ordering

UPDATE @eHubRoutingRule
SET RR_LastUpdateUTC = NULL

declare @XML1 xml
declare @XML2 xml
declare @XML3 xml

SET @XML1 = (SELECT * FROM eHubClient WHERE CC_PK = @FPM_PK OR CC_PK IN (SELECT RR_Success_CC_Recipient FROM @eHubRoutingRule WHERE RR_Success_CC_Recipient IS NOT NULL) for xml path ('eHubClient'))
SET @XML2 = (SELECT * FROM eHubRoutingRuleFact WHERE RX_RR = @RoutingRuleRoot for xml path ('eHubRoutingRuleFact'))
SET @XML3 = (SELECT * FROM @eHubRoutingRule for xml path ('eHubRoutingRule'))

DECLARE @siteMapXml XML 

SET @siteMapXml = (SELECT @XML1, @XML2, @XML3 FOR XML PATH(''))

SELECT '<?xml version="1.0" encoding="utf-8" ?><eHubTransactions xmlns="http://tempuri.org/eHubTransactions.xsd"><!-- Autogenerated from database server -->' + CAST(@siteMapXml as nvarchar(max)) + '</eHubTransactions>'

2/ Query to generate for single provider (Please update your [PROVIDERNAME]/[TESTCLASSNAME]Data/eHubTransactions_Infrastructure.xml):

-- Modifiable
declare @Provider nvarchar(25) = 'NGBEDI'

-- QUERY
DECLARE @FPM_PK uniqueidentifier = '44A7E26D-F2B7-4905-81F0-40BB64D4CE1B'
DECLARE @ProviderPK uniqueidentifier
SELECT @ProviderPK=CC_PK FROM eHubClient WHERE CC_ID = @Provider

DECLARE @eHubRoutingRule TABLE
(
      [RR_PK] [uniqueidentifier] NOT NULL,
      [RR_Condition_Expression] [nvarchar](max) NULL,
      [RR_Group_RR_GroupRule] [uniqueidentifier] NULL,
      [RR_Group_MatchMultiple] [bit] NULL,
      [RR_Success_CC_Recipient] [uniqueidentifier] NULL,
      [RR_Success_RR_SubRule] [uniqueidentifier] NULL,
      [RR_Failed_ErrorCode] [nvarchar](50) NULL,
      [RR_Failed_ErrorDescription] [nvarchar](2000) NULL,
      [RR_Failed_RR_SubRule] [uniqueidentifier] NULL,
      [RR_Result_Value] [nvarchar](2000) NULL,
      [RR_Success_SP_Provider] [uniqueidentifier] NULL,
      [RR_LastUpdateUTC] [datetime2](7) NULL,
      [RR_Group_Ordering] [int] NULL
)
;WITH x (RR_PK,    RR_Condition_Expression,      RR_Group_RR_GroupRule,      RR_Group_MatchMultiple,      RR_Success_CC_Recipient,      RR_Success_RR_SubRule,   RR_Failed_ErrorCode,      RR_Failed_ErrorDescription,      RR_Failed_RR_SubRule,    RR_Result_Value,      RR_Success_SP_Provider,  RR_LastUpdateUTC,      RR_Group_Ordering)
AS
(
   select RR_PK,   RR_Condition_Expression,      RR_Group_RR_GroupRule,      RR_Group_MatchMultiple,      RR_Success_CC_Recipient,      RR_Success_RR_SubRule,   RR_Failed_ErrorCode,      RR_Failed_ErrorDescription,      RR_Failed_RR_SubRule,    RR_Result_Value,      RR_Success_SP_Provider,  RR_LastUpdateUTC,      RR_Group_Ordering
   from eHubRoutingRule
   where RR_PK IN (SELECT SP_RR FROM eHubServiceProvider WHERE SP_CC_Service = @FPM_PK AND SP_CC_Provider = @ProviderPK)
   UNION ALL
   select t.RR_PK, t.RR_Condition_Expression,      t.RR_Group_RR_GroupRule,      t.RR_Group_MatchMultiple,      t.RR_Success_CC_Recipient,      t.RR_Success_RR_SubRule,      t.RR_Failed_ErrorCode,      t.RR_Failed_ErrorDescription,      t.RR_Failed_RR_SubRule,  t.RR_Result_Value,      t.RR_Success_SP_Provider, t.RR_LastUpdateUTC,      t.RR_Group_Ordering
   from eHubRoutingRule as t inner join
       x as parent on t.RR_Group_RR_GroupRule = parent.RR_PK OR t.RR_PK = parent.RR_Success_RR_SubRule OR t.RR_PK = parent.RR_Failed_RR_SubRule
)
insert into @eHubRoutingRule SELECT * from x

UPDATE @eHubRoutingRule
SET RR_LastUpdateUTC = NULL

declare @XML1 xml
declare @XML2 xml
declare @XML3 xml

SET @XML1 = (SELECT * FROM eHubClient WHERE CC_ID = @Provider OR CC_PK IN (SELECT RR_Success_CC_Recipient FROM @eHubRoutingRule WHERE RR_Success_CC_Recipient IS NOT NULL) for xml path ('eHubClient'))
SET @XML2 = (SELECT * FROM eHubServiceProvider WHERE SP_CC_Service = @FPM_PK AND SP_CC_Provider = @ProviderPK for xml path ('eHubServiceProvider'))
SET @XML3 = (SELECT * FROM @eHubRoutingRule for xml path ('eHubRoutingRule'))


DECLARE @siteMapXml XML 

SET @siteMapXml = (SELECT @XML1, @XML2, @XML3 FOR XML PATH(''))

SELECT '<?xml version="1.0" encoding="utf-8" ?><eHubTransactions xmlns="http://tempuri.org/eHubTransactions.xsd"><!-- Autogenerated from database server -->' + CAST(@siteMapXml as nvarchar(max)) + '</eHubTransactions>'
